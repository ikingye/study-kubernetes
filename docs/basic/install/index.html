                                    Kubernetes 各组件配置文件 #

                                    kubelet #

                                    kubelet 是通过 systemctl 来管理的，因此配置文件在 /etc/systemd/system/ 或 /usr/lib/systemd/system/

                                    # 修改配置文件
vi /usr/lib/systemd/system/kubelet.service.d/10-kubeadm.conf

# kubeadm init 生成的
# 使用 systemd
vi /var/lib/kubelet/config.yaml
# 配置 6443
vi /etc/kubernetes/kubelet.conf

# 重启
systemctl daemon-reload && systemctl restart kubelet

# 查看 kubelet 错误日志
journalctl -xef | egrep ": [F][0-9]" -B 1

                                    kubeadm vs kops vs kubespray #

                                    • Kubespray 基于 Ansible 实现配置和编排
                                      • 更灵活，适用多平台
                                    • Kops 自己实现配置和编排
                                      • 适用某个固定平台
                                    • kubeadm
                                      • 需要领域知识，类似 Kubernetes cluster operator
                                      • 从 v2.3 开始，kubespray 底层使用 kubeadm
